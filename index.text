        function navigateTo(route, pushState = true) {
            if (isTransitioning || route === currentRoute) return;
            isTransitioning = true;

            // Update URL hash
            if (pushState) {
                window.location.hash = route;
            }

            // Hide all pages
            document.querySelectorAll('.page').forEach(el => el.classList.remove('active'));

            // Show target
            const target = document.getElementById(pageMap[route]);
            if (target) {
                target.classList.add('active');
                // Re-trigger animation
                const content = target.querySelector('.page-content');
                if (content) {
                    content.classList.remove('exit');
                    // Force reflow
                    void content.offsetWidth;
                    content.style.animation = 'none';
                    void content.offsetWidth;
                    content.style.animation = '';
                }
            }

            // Update nav links
            document.querySelectorAll('.nav-links a, .mobile-nav a').forEach(el => {
                el.classList.toggle('active', el.dataset.route === route);
            });

            currentRoute = route;
            isTransitioning = false;

            // Trigger page-specific animations
            if (route === 'home') initHomeStats();
            if (route === 'myApp') initProjectCards();
            if (route === 'about') initAboutObserver();
            if (route === 'contact') initContactCards();
        }

            initContactCards();

            initRouteEvents();

            // Trigger stats if on home
            if (startRoute === 'home') {
                setTimeout(initHomeStats, 300);
            }
        }

        function initTilt() {
            const wrapper = document.getElementById('tiltWrapper');
            if (!wrapper) return;

            let isTilting = false;

            wrapper.addEventListener('mousemove', function(e) {
                const rect = this.getBoundingClientRect();
                const x = (e.clientX - rect.left) / rect.width - 0.5;
                const y = (e.clientY - rect.top) / rect.height - 0.5;
                this.style.transform = 'rotateY(' + (x * 8) + 'deg) rotateX(' + (y * -8) + 'deg)';
                isTilting = true;
            });

            wrapper.addEventListener('mouseleave', function() {
                this.style.transform = 'rotateY(0deg) rotateX(0deg)';
                isTilting = false;
            });
        }

        let statsAnimated = false;

        function initHomeStats() {
            const container = document.getElementById('statsContainer');
            if (!container) return;
            if (statsAnimated) return;

            const observer = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting && !statsAnimated) {
                        animateStats();
                        statsAnimated = true;
                        observer.disconnect();
                    }
                });
            }, { threshold: 0.3 });

            observer.observe(container);
        }

        function animateStats() {
            const target = { projects: 20, featured: 7, satisfaction: 100 };
            let progress = 0;
            const duration = 1200;
            const step = 20;

            const elProjects = document.getElementById('statProjects');
            const elFeatured = document.getElementById('statFeatured');
            const elSatisfaction = document.getElementById('statSatisfaction');

            const interval = setInterval(() => {
                progress += step;
                const p = Math.min(progress / duration, 1);
                elProjects.textContent = Math.floor(p * target.projects);
                elFeatured.textContent = Math.floor(p * target.featured);
                elSatisfaction.textContent = Math.floor(p * target.satisfaction);
                if (p >= 1) clearInterval(interval);
            }, 30);
        }

        function initProjectCards() {
            const cards = document.querySelectorAll('#featuredGrid .project-card');
            if (!cards.length) return;

            const observer = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        const el = entry.target;
                        const idx = parseInt(el.dataset.index) || 0;
                        const delay = idx * 0.05;
                        setTimeout(() => {
                            el.style.transition = 'opacity 0.4s ease, transform 0.4s ease';
                            el.style.opacity = '1';
                            el.style.transform = 'translateY(0)';
                        }, delay * 1000);
                        observer.unobserve(el);
                    }
                });
            }, { threshold: 0.1 });

            cards.forEach(card => observer.observe(card));
        }

        let currentModalId = null;

        function initModalEvents() {
            // Delegate clicks on overview buttons
            document.addEventListener('click', function(e) {
                const btn = e.target.closest('.btn-overview');
                if (btn) {
                    const id = btn.dataset.projectId;
                    if (id) openModal(id);
                }
            });

            // Close modal
            document.getElementById('modalCloseBtn').addEventListener('click', closeModal);
            document.getElementById('modalOverlay').addEventListener('click', function(e) {
                if (e.target === this) closeModal();
            });

            // Escape key
            document.addEventListener('keydown', function(e) {
                if (e.key === 'Escape') closeModal();
            });
        }

        function openModal(id) {
            const project = featuredProjects.find(p => p.id === id);
            if (!project) return;
            currentModalId = id;

            const overlay = document.getElementById('modalOverlay');
            document.getElementById('modalTitle').textContent = project.title;
            document.getElementById('modalCategory').textContent = project.category;

            const body = document.getElementById('modalBody');
            body.innerHTML = `
            <div class="modal-section">
              <h4>Problem</h4>
              <p>${project.overview.problem}</p>
            </div>
            <div class="modal-section">
              <h4>Solution</h4>
              <p>${project.overview.solution}</p>
            </div>
            <div class="modal-section">
              <h4>Impact</h4>
              <p>${project.overview.impact}</p>
            </div>
            <div class="modal-section">
              <h4>Tech Stack</h4>
              <div class="modal-tech">${project.tech.map(t => `<span>${t}</span>`).join('')}</div>
            </div>
          `;

            const actions = document.getElementById('modalActions');
            actions.innerHTML = `
            <a href="${project.live}" target="_blank" rel="noopener" class="btn-primary">Live Demo</a>
            <a href="${project.github}" target="_blank" rel="noopener" class="btn-secondary">GitHub</a>
          `;

            // Prevent body scroll
            document.body.style.overflow = 'hidden';

            // Open with animation
            requestAnimationFrame(() => {
                overlay.classList.add('open');
            });
        }

        function closeModal() {
            const overlay = document.getElementById('modalOverlay');
            overlay.classList.remove('open');
            document.body.style.overflow = '';
            currentModalId = null;
        }

        let aboutAnimated = false;

        function initAboutObserver() {
            const section = document.getElementById('aboutSection');
            if (!section) return;
            if (aboutAnimated) return;

            const observer = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting && !aboutAnimated) {
                        animateAbout(section);
                        aboutAnimated = true;
                        observer.disconnect();
                    }
                });
            }, { threshold: 0.1 });

            observer.observe(section);
        }

        function animateAbout(container) {
            const children = container.querySelectorAll('.about-content, .journey, .tech-section, .resume-section');
            children.forEach((el, i) => {
                const delay = i * 0.08;
                el.style.opacity = '0';
                el.style.transform = 'translateY(16px)';
                el.style.transition = 'opacity 0.4s ease, transform 0.4s ease';
                setTimeout(() => {
                    el.style.opacity = '1';
                    el.style.transform = 'translateY(0)';
                }, delay * 1000 + 100);
            });
        }

        // ============================================================
        // NAV EVENTS
        // ============================================================

        function initNavEvents() {
            // All route links
            document.querySelectorAll('[data-route]').forEach(el => {
                el.addEventListener('click', function(e) {
                    e.preventDefault();
                    const route = this.dataset.route;
                    if (route) {
                        navigateTo(route);
                        closeMobileMenu();
                    }
                });
            });

            // Hamburger
            document.getElementById('hamburgerBtn').addEventListener('click', function() {
                toggleMobileMenu();
            });

            // Overlay close
            document.getElementById('mobileOverlay').addEventListener('click', closeMobileMenu);

            // Mobile close button
            document.getElementById('mobileCloseBtn').addEventListener('click', closeMobileMenu);

            // Hash change
            window.addEventListener('hashchange', function() {
                const hash = window.location.hash.replace('#', '');
                if (routes.includes(hash) && hash !== currentRoute) {
                    navigateTo(hash, false);
                }
            });
        }

        function toggleMobileMenu() {
            const isOpen = document.getElementById('mobileNav').classList.contains('open');
            if (isOpen) closeMobileMenu();
            else openMobileMenu();
        }

        function openMobileMenu() {
            document.getElementById('mobileNav').classList.add('open');
            document.getElementById('mobileOverlay').classList.add('open');
            document.getElementById('hamburgerBtn').classList.add('open');
            document.body.style.overflow = 'hidden';
        }

        function closeMobileMenu() {
            document.getElementById('mobileNav').classList.remove('open');
            document.getElementById('mobileOverlay').classList.remove('open');
            document.getElementById('hamburgerBtn').classList.remove('open');
            document.body.style.overflow = '';
        }

        // ============================================================
        // ROUTE EVENTS (for page-specific init)
        // ============================================================

        function initRouteEvents() {
            // Re-init when page becomes visible via nav
            // We use a MutationObserver or just call on navigation
            // For simplicity, we'll re-init on each navigation
            const origNavigate = navigateTo;
            navigateTo = function(route, pushState) {
                origNavigate.call(this, route, pushState);
                // Re-trigger page-specific animations after transition
                setTimeout(() => {
                    if (route === 'home') {
                        if (!statsAnimated) initHomeStats();
                    }
                    if (route === 'myApp') initProjectCards();
                    if (route === 'about') initAboutObserver();
                    if (route === 'contact') initContactCards();
                }, 400);
            };
        }
